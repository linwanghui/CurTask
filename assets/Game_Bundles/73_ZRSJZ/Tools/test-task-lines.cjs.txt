function load(file) {
    file=path.resolve(root,file);
    if(cache.has(file))return cache.get(file).exports;
    const mod={exports:{}};cache.set(file,mod);
    const src=ts.transpileModule(fs.readFileSync(file,'utf8'),{compilerOptions:{target:ts.ScriptTarget.ES2020,module:ts.ModuleKind.CommonJS,experimentalDecorators:true}}).outputText;
    const localRequire=req=> {
        if(stubs[req])return stubs[req];
        const base=path.basename(req);
        if(stubs[base])return stubs[base];
        return load(path.resolve(path.dirname(file),req+'.ts'));
    };
    vm.runInThisContext('(function(require,module,exports){'+src+'\n})',{filename:file})(localRequire,mod,mod.exports);
    return mod.exports;
}
