function load(file, requireMock = () => { throw Error('Unexpected dependency'); }) {
    const exports = {};
    const result = ts.transpileModule(fs.readFileSync(file, 'utf8'), { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2019, experimentalDecorators: true } });
    vm.runInNewContext(result.outputText, { exports, require: requireMock, console });
    return exports;
}

async function testPurchase() {
    const data = { Gold: 1000 };
    let deliveries = 0, finish;
    const ui = { ShowTip: () => {}, ReceivePropAwards: () => { deliveries++; return new Promise(resolve => { finish = resolve; }); } };
    const cc = { _decorator: { ccclass: () => type => type, property: () => () => {} }, Enum: x => x, isValid: () => true };
    const panelClass = load('Scripts/Panel/ZRSJZ_MysteriousMerchantPanel.ts', name => {
        if (name === 'cc') return cc;
        if (name.endsWith('/ZRSJZ_Panel')) return { ZRSJZ_Panel: class {} };
        if (name.includes('UIManager')) return { ZRSJZ_UIManager: { Instance: ui, ZRSJZ_DLC: true } };
        if (name.includes('AudioManager')) return { ZRSJZ_AudioManager: { Instance: { PlaySound: () => {} } } };
        if (name.includes('MerchantService')) return { ZRSJZ_MerchantService: merchant };
        if (name.includes('AccountService')) return { ZRSJZ_AccountService: { ChangeGold: amount => { data.Gold += amount; } } };
        if (name.includes('GameData')) return { ZRSJZ_GameData: { Instance: data } };
        return constants;
    }).ZRSJZ_MysteriousMerchantPanel;
    const panel = new panelClass();
    panel.Render = () => {};
    merchant.Visible = true;
    merchant.Goods = [{ Name: '一级头', Price: 100, Count: 1, Sold: false, Buying: false }];
    data.Gold = 99;
    await panel.Buy(0);
    assert.equal(deliveries, 0);
    assert.equal(data.Gold, 99);
    data.Gold = 1000;
    const first = panel.Buy(0);
    await panel.Buy(0);
    assert.equal(deliveries, 1);
    assert.equal(data.Gold, 900);
    finish({ InvalidAwards: [], MailID: 'overflow-mail' });
    await first;
    assert.equal(merchant.Goods[0].Sold, true);
    await panel.Buy(0);
    assert.equal(deliveries, 1);
    assert.equal(data.Gold, 900);
    console.log('PASS: insufficient gold, concurrent clicks, single charge, mail overflow result and sold-out repurchase guard.');
}
testPurchase().catch(error => { console.error(error); process.exitCode = 1; });
