const fs = require('fs');
const vm = require('vm');
const assert = require('assert');
const ts = require('D:/CocosCreator/Creator/3.8.6/resources/app.asar.unpacked/node_modules/typescript/lib/typescript.js');
let saves = 0;
const data = { SupplyOffer: -1, SupplyLastClaimTime: 0, SupplyFailurePending: false };
const gameData = { Instance: data, SaveData: () => saves++ };
const source = fs.readFileSync('Scripts/Service/ZRSJZ_ActionSuppliesService.ts', 'utf8');
const exportsObject = {};
vm.runInNewContext(ts.transpileModule(source, { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2019 } }).outputText,
    { exports: exportsObject, require: () => ({ ZRSJZ_GameData: gameData }), Date, Math });
const service = exportsObject.ZRSJZ_ActionSuppliesService;
const start = 100000000;

assert.equal(saves, 0);
assert.equal(service.Refresh(true, start, () => 0.2), true);
assert.equal(data.SupplyOffer, 0);
assert.equal(service.GetAwards().length, 6);
assert.deepEqual(Array.from(service.GetAwards().slice(4), x => x.Count), [60, 60]);
const initial = JSON.stringify(service.GetAwards());
service.Refresh(true, start + 1, () => { throw Error('Reopening must not reroll'); });
assert.equal(JSON.stringify(service.GetAwards()), initial);
service.CompleteClaim(start);
assert.equal(service.GetAwards().length, 0);
assert.equal(service.Refresh(true, start + service.Cooldown - 1), false);
service.MarkFailureReturn();
assert.equal(service.Refresh(true, start + 1000, () => 0.5), false);
assert.equal(data.SupplyFailurePending, false);
service.Refresh(true, start + 2000, () => { throw Error('Failure must only roll once'); });
service.MarkFailureReturn();
assert.equal(service.Refresh(false, start + 3000), false);
assert.equal(data.SupplyFailurePending, true);
assert.equal(service.Refresh(true, start + 3000, () => 0.499), true);
service.CompleteClaim(start + 4000);
assert.equal(service.Refresh(true, start + 4000 + service.Cooldown, () => 0.9), true);
assert.equal(data.SupplyOffer, 1);
const constant = fs.readFileSync('Scripts/ZRSJZ_Constant.ts', 'utf8');
for (const variant of [0, 1]) {

    for (const award of service.GetAwards()) assert(constant.includes('Name: "' + award.PropName + '"'));
}
const prefab = JSON.parse(fs.readFileSync('../73_ZRSJZ_DLC/Prefabs/Panel/行动补给弹窗.prefab', 'utf8'));
function verify(value) {
    if (!value || typeof value !== 'object') return;
    if ('__id__' in value) assert(prefab[value.__id__], 'Invalid object reference: ' + value.__id__);
    Object.values(value).forEach(verify);
}
verify(prefab);
const panel = prefab.find(x => Array.isArray(x.RewardSlots));
assert.equal(panel.RewardSlots.length, 6);
for (const [i, ref] of panel.RewardSlots.entries()) {
    const slot = prefab[ref.__id__];
    assert.deepEqual(slot._children.map(r => prefab[r.__id__]._name), ['图标', '名字', '数量']);
    const button = slot._components.map(r => prefab[r.__id__]).find(x => x.__type__ === 'cc.Button');
    const event = prefab[button.clickEvents[0].__id__];
    assert.equal(event.handler, 'OnRewardClick');
    assert.equal(event.customEventData, String(i));
}
const scene = JSON.parse(fs.readFileSync('Scenes/ZRSJZ_Start.scene', 'utf8'));

console.log('PASS: DLC gating, persisted offers, cooldown boundaries, failure rolls, reward config, prefab references and click bindings.');
